void HAL_TIM_PeriodElapsedCallback(TIM_HandleTypeDef *htim)
{
    dot_counter--;
    if (dot_counter <= 0)
    {
        dot_counter = 100;
        HAL_GPIO_TogglePin(DOT_GPIO_Port, DOT_Pin);
    }

    digit_counter--;
    if (digit_counter <= 0)
    {
        digit_counter = 50;
        update7SEG(index_led++);
        HAL_GPIO_TogglePin(LED_GPIO_Port, LED_Pin);


        if (index_led >= MAX_LED)
        	index_led = 0;
    }
}
